{{/* CJK-aware word count + reading time. Consumes a Page context. Usage: {{ partial "cjk-info.html" . }} Renders the article-header two-part stats: "全文约 N 字 · 预计阅读 M 分钟". To render compact "N 字 · M 分钟" form, pass a dict with compact=true. */}} {{ $compact := false }} {{ $ctx := . }} {{ if (hasPrefix (printf "%T" .) "map") }}{{ $ctx = .page }}{{ $compact = .compact | default false }}{{ end }} {{ $plain := $ctx.Plain }} {{ $han := len (findRE `[\p{Han}]` $plain) }} {{ $alnum := len (findRE `[A-Za-z0-9]+` $plain) }} {{ $words := add $han $alnum }} {{ $wpm := $ctx.Site.Params.cjkWordsPerMinute | default 400 }} {{ $minutes := div (add $words (sub $wpm 1)) $wpm }} {{ if lt $minutes 1 }}{{ $minutes = 1 }}{{ end }} {{/* thousands separator */}} {{ $num := printf "%d" $words }} {{ $grouped := $num }} {{ if gt $words 999 }} {{ $len := len $num }} {{ $lead := mod $len 3 }} {{ if eq $lead 0 }}{{ $lead = 3 }}{{ end }} {{ $grouped = "" }} {{ range $i := seq 0 (sub $len 1) }} {{ $grouped = printf "%s%s" $grouped (substr $num $i 1) }} {{ $p := add $i 1 }} {{ if and (ge $p $lead) (lt $p $len) (eq (mod (sub $p $lead) 3) 0) }}{{ $grouped = printf "%s," $grouped }}{{ end }} {{ end }} {{ end }} {{ if $compact }} {{ $grouped }} 字 · {{ $minutes }} 分钟 {{ else }} 全文约 {{ $grouped }} 字·预计阅读 {{ $minutes }} 分钟 {{ end }}
